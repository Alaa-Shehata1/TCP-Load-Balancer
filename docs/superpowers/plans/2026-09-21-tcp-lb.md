# TCP Load Balancer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Go L4 TCP load balancer with round-robin + least-connections, health checks, failover, and observability, demoable via Docker Compose.

**Architecture:** Single Go binary `cmd/lb` fronts 3 Go echo backends. Pluggable `Balancer` interface picks a healthy backend from `Pool`; `Proxy` dials with timeout and `io.Copy` both directions; `Healthchecker` actively dials + passively ejects; admin server exposes `/health` + Prometheus `/metrics`.

**Tech Stack:** Go 1.22+, `gopkg.in/yaml.v3`, `prometheus/client_golang`, Docker Compose, `slog`, `go test -race`, `golangci-lint`

**Spec:** `docs/specs/2026-09-21-tcp-lb-design.md`

## Global Constraints
- Go >= 1.22, `gofmt` clean on every commit.
- `go vet ./...` must pass before commit.
- All blocking ops take `context.Context` where applicable; timeouts from config, defaults dial 3s / idle 60s / HC interval 5s.
- Errors wrapped with `fmt.Errorf("...: %w", err)`, no ignored errors.
- Table-driven tests + `go test -race ./...`.
- Backend choice pinned at connection start, never migrated mid-connection.
- Pipes via `io.Copy`, no full-stream buffering, `allowHalfOpen=false` semantics (close both sides on error).

## Review Focus
- Client connects when zero backends healthy → expect clean RST/close + log + `connections_total{result="no_healthy"}` increment, not hang.
- Backend accepts then stalls past idle timeout → expect idle deadline closes both sides, LeastConn counter decremented.
- Backend dies mid-stream → expect both sockets destroyed, no FD leak, HC marks unhealthy within 2 intervals.
- `docker stop server-b` during 100 concurrent conns → expect new conns only to A/C, no panic in RR counter overflow.
- Malformed `config.yaml` (bad port, unknown algorithm) → expect startup fail-fast with explicit error, exit non-zero.

```go
// Package config loads and validates YAML for the LB.
package config

import (
    "fmt"
    "net"
    "os"

    "gopkg.in/yaml.v3"
)

type BackendConfig struct {
    Name string `yaml:"name"`
    Host string `yaml:"host"`
    Port int    `yaml:"port"`
    Addr string `yaml:"-"`
}
type HCConfig struct {
    IntervalSecs int `yaml:"interval_secs"`
    TimeoutSecs  int `yaml:"timeout_secs"`
    FailThreshold int `yaml:"fail_threshold"`
}
type TimeoutConfig struct {
    DialSecs int `yaml:"dial_secs"`
    IdleSecs int `yaml:"idle_secs"`
}
type Config struct {
    ListenAddr  string          `yaml:"listen_addr"`
    AdminAddr   string          `yaml:"admin_addr"`
    Algorithm   string          `yaml:"algorithm"`
    Backends    []BackendConfig `yaml:"backends"`
    Healthcheck HCConfig        `yaml:"healthcheck"`
    Timeouts    TimeoutConfig   `yaml:"timeouts"`
}

func Load(path string) (Config, error) {
    var c Config
    b, err := os.ReadFile(path)
    if err != nil { return c, fmt.Errorf("read config %s: %w", path, err) }
    if err := yaml.Unmarshal(b, &c); err != nil { return c, fmt.Errorf("parse config: %w", err) }
    if c.Algorithm != "round-robin" && c.Algorithm != "least-conn" {
        return c, fmt.Errorf("unknown algorithm %q", c.Algorithm)
    }
    if len(c.Backends) == 0 { return c, fmt.Errorf("no backends") }
    for i := range c.Backends {
        bc := &c.Backends[i]
        if bc.Port <= 0 || bc.Port > 65535 { return c, fmt.Errorf("backend %s bad port %d", bc.Name, bc.Port) }
        bc.Addr = net.JoinHostPort(bc.Host, fmt.Sprint(bc.Port))
    }
    if c.ListenAddr == "" { c.ListenAddr = ":9000" }
    if c.AdminAddr == "" { c.AdminAddr = ":8080" }
    if c.Healthcheck.IntervalSecs == 0 { c.Healthcheck.IntervalSecs = 5 }
    if c.Healthcheck.TimeoutSecs == 0 { c.Healthcheck.TimeoutSecs = 2 }
    if c.Healthcheck.FailThreshold == 0 { c.Healthcheck.FailThreshold = 2 }
    if c.Timeouts.DialSecs == 0 { c.Timeouts.DialSecs = 3 }
    if c.Timeouts.IdleSecs == 0 { c.Timeouts.IdleSecs = 60 }
    return c, nil
}
```

```go
// internal/balancer/balancer.go
package balancer

import (
    "fmt"
    "sync/atomic"
    "github.com/alaa157/tcp-load-balancer/internal/pool"
)

type Balancer interface{ Next() *pool.Backend }

type roundRobin struct {
    p *pool.Pool
    n atomic.Uint64
}
func (r *roundRobin) Next() *pool.Backend {
    h := r.p.Healthy()
    if len(h)==0 { return nil }
    i := r.n.Add(1)-1
    return h[int(i%uint64(len(h)))]
}

type leastConn struct{ p *pool.Pool }
func (l *leastConn) Next() *pool.Backend {
    h := l.p.Healthy()
    if len(h)==0 { return nil }
    best := h[0]
    for _, b := range h[1:] {
        if b.Active() < best.Active() { best = b }
    }
    return best
}

func New(algo string, p *pool.Pool) (Balancer, error) {
    switch algo {
    case "round-robin": return &roundRobin{p:p}, nil
    case "least-conn": return &leastConn{p:p}, nil
    default: return nil, fmt.Errorf("unknown algorithm %q", algo)
    }
}
```

Wire `Active() int64`, `AddConn/DoneConn` on Backend with atomics.

```go
package proxy

import (
    "context"
    "fmt"
    "io"
    "log/slog"
    "net"
    "sync/atomic"
    "time"

    "github.com/alaa157/tcp-load-balancer/internal/balancer"
    "github.com/alaa157/tcp-load-balancer/internal/pool"
)

type Server struct {
    p *pool.Pool
    b balancer.Balancer
    dialTimeout time.Duration
    idleTimeout time.Duration
    log *slog.Logger
    conns atomic.Uint64
}

func New(p *pool.Pool, b balancer.Balancer, dial, idle time.Duration, log *slog.Logger) *Server {
    return &Server{p:p, b:b, dialTimeout:dial, idleTimeout:idle, log:log}
}

func (s *Server) Serve(l net.Listener) error {
    for {
        c, err := l.Accept()
        if err != nil { return fmt.Errorf("accept: %w", err) }
        go s.HandleConn(c)
    }
}

func (s *Server) HandleConn(client net.Conn) {
    id := s.conns.Add(1)
    be := s.b.Next()
    if be == nil {
        s.log.Warn("no healthy backend", "conn", id)
        client.Close()
        return
    }
    d := &net.Dialer{Timeout: s.dialTimeout}
    up, err := d.DialContext(context.Background(), "tcp", be.Addr)
    if err != nil {
        s.log.Warn("dial failed, retry next", "backend", be.Addr, "err", err)
        s.p.MarkUnhealthy(be.Addr)
        be2 := s.b.Next()
        if be2 == nil || be2.Addr == be.Addr { client.Close(); return }
        up, err = d.DialContext(context.Background(), "tcp", be2.Addr)
        if err != nil { s.p.MarkUnhealthy(be2.Addr); client.Close(); return }
        be = be2
    }
    s.p.AddConn(be.Addr)
    defer s.p.DoneConn(be.Addr)
    s.log.Info("proxy", "conn", id, "backend", be.Addr)
    _ = client.SetDeadline(time.Now().Add(s.idleTimeout))
    _ = up.SetDeadline(time.Now().Add(s.idleTimeout))
    go func(){ _, _ = io.Copy(up, client); _ = up.(*net.TCPConn).CloseWrite() }()
    _, _ = io.Copy(client, up)
    client.Close(); up.Close()
}
```
